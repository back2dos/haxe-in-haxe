# HXB conformance of upstream test suites beyond tests/misc/eval and tests/optimization
# (test/conformance-cases.txt): the configurations of upstream CI's macro job
# (haxe/tests/runci/targets/Macro.hx), compiled to HXB instead of being run.
# Most create a macro context and use utest (-lib utest), which conformance.py
# finds in the haxelib repository of test/upstream.py (HAXELIB_PATH); set it up
# with `python3 test/upstream.py --reference <any step>` first. Run with:
#   HAXE_REFERENCE=haxe/haxe python3 test/conformance.py --cases test/conformance-upstream-cases.txt [--hxb-lib]
# Format: see conformance-cases.txt (hxml cases pass their remaining arguments after the hxml).
#   volatile <case> <member> <reason>: the member's payload embeds the time of compilation;
#   it must exist in both archives, but its bytes are not compared.
# Exclusions are recorded in upstream-hxb-exclusions.md ("Upstream suites").

# tests/unit: the whole unit suite (unit-macro step without its --hxb argument)
hxml unit-macro haxe/tests/unit compile-macro.hxml
# HelperMacros.getCompilationDate embeds Std.string(Date.now()) (to the second)
volatile unit-macro eval/unit/TestMainNow.hxb compile-time date
# the same without dead code elimination: every field of every module the suite loads
hxml unit-macro-dce-no haxe/tests/unit compile-macro.hxml -dce no
volatile unit-macro-dce-no eval/unit/TestMainNow.hxb compile-time date

# tests/nullsafety (nullsafety step)
hxml nullsafety haxe/tests/nullsafety test.hxml

# tests/threads (threads step)
hxml threads haxe/tests/threads build.hxml --interp

# tests/sys: the four parts of compile-eval-hxb.hxml (compile-each.hxml without its
# `--cmd haxe compile-neko-exitcode.hxml` part, which builds a Neko binary)

hxml sys-ExitCode haxe/tests/sys -D source-header= --debug -lib utest -p src compile-fs.hxml ExitCode --interp
hxml sys-UtilityProcess haxe/tests/sys -D source-header= --debug -lib utest -p src compile-fs.hxml UtilityProcess --interp
